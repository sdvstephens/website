\begin{tikzpicture}[font=\small]

  \node[draw, thick] at (2.5,-1.05) {Poisson distribution};

\begin{scope}[xshift=7.6cm]

  \node[rotate=90, above] at (-0.05,1.7) {nodes with $k$ links};
  \node[align=left, anchor=west] (b) at (0.6,2.7) {large number of nodes\\ have few links};

  \node[align=left, anchor=west] (c) at (2.4,1.2) {small number of nodes\\ (hubs) have many links};
