\begin{tikzpicture}[
  box/.style={draw, rounded corners, thick, align=center, font=\small, minimum height=1cm, inner sep=4pt},

  arr/.style={-{Stealth[length=2.5mm]}, thick},
  hl/.style={font=\scriptsize, midway}
]
% ---- row 1: the HPA axis ----
\node[box, draw=none, font=\small\bfseries] (stress) at (0,0) {Stress};
\node[box] (hyp) at (2.6,0) {Hypothalamus};
\node[box] (pit) at (5.6,0) {Pituitary\\gland};
\node[box] (adr) at (8.4,0) {Adrenal\\gland};

\node[stp, text width=4.4cm] (brk) at (6.4,-2.6) {Breakdown of neurons \& synapses, especially GABA projections from PFC \& hippocampus to amygdala};
