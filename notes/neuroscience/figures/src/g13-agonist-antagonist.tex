\begin{tikzpicture}[
  font=\small,
  cell/.style={draw=blue, very thick, fill=blue!8, rounded corners=6pt},
  rec/.style={draw=orange, thick, fill=orange!35},
  ago/.style={circle, draw=green!60!black, thick, fill=green!45, minimum size=0.55cm, inner sep=0pt},
  sig/.style={-{Stealth[length=2mm]}, line width=1.3pt, purple},
  move/.style={-{Stealth[length=2mm]}, thick},
  ttl/.style={font=\small\bfseries, align=center},
  outl/.style={font=\small, align=center}
]
% a cell with a receptor (a cup opening upward) on top, centered at x = #1
\newcommand{\cellat}[1]{%
  \draw[cell] (#1-1.1,-2.6) rectangle (#1+1.1,0);
  \draw[rec] (#1-0.5,-0.25) -- (#1-0.5,0.55) -- (#1-0.25,0.55) -- (#1-0.25,0.2)
    -- (#1+0.25,0.2) -- (#1+0.25,0.55) -- (#1+0.5,0.55) -- (#1+0.5,-0.25) -- cycle;
}
% antagonist: a T-shaped block whose stem fits the receptor cup
\newcommand{\antag}[3]{% x, y, rotation
  \begin{scope}[shift={(#1,#2)}, rotate=#3]
    \fill[red!70, draw=red, thick] (-0.42,0.42) -- (0.42,0.42) -- (0.42,0.18) -- (0.14,0.18)
      -- (0.14,-0.18) -- (-0.14,-0.18) -- (-0.14,0.18) -- (-0.42,0.18) -- cycle;
  \end{scope}
}

% ---------- 1. agonist alone ----------
\node[ttl] at (0,3.3) {Agonist alone};
\cellat{0}
\node[ago] (a0) at (0,2.25) {};
\node[ago] at (0,0.55) {};
\draw[move, green!50!black] (a0) -- (0,1.0);
\foreach \x in {-0.3,0,0.3} \draw[sig] (\x,-0.35) -- (\x,-0.9);
\node[star, star points=12, star point ratio=1.9, minimum size=1.15cm, fill=purple!45, draw=purple] at (0,-1.65) {};
\draw[-{Stealth[length=3mm]}, line width=3pt, purple] (0,-2.7) -- (0,-3.3);
\node[outl] at (0,-3.65) {Full activation};

% ---------- 2. agonist + antagonist ----------
\begin{scope}[xshift=3.6cm]
\node[ttl] at (0,3.3) {Agonist + antagonist};
\cellat{0}
\node[ago] (b0) at (-1.0,2.2) {};
\node[ago] at (-0.38,0.85) {};
\draw[move, green!50!black] (b0) -- (-0.5,1.15);
\antag{0.32}{0.78}{-35}
\antag{1.05}{2.15}{-35}
\draw[move, red] (0.85,1.75) -- (0.5,1.2);
\draw[sig] (0,-0.35) -- (0,-0.9);
\node[star, star points=10, star point ratio=1.9, minimum size=0.7cm, fill=purple!45, draw=purple] at (0,-1.6) {};
\draw[-{Stealth[length=2.5mm]}, line width=1.8pt, purple] (0,-2.7) -- (0,-3.3);
\node[outl] at (0,-3.65) {Less activation};
\end{scope}

% ---------- 3. antagonist alone ----------
\begin{scope}[xshift=7.2cm]
\node[ttl] at (0,3.3) {Antagonist alone};
\cellat{0}
\antag{0}{2.2}{0}
\draw[move, red] (0,1.7) -- (0,0.95);
\antag{0}{0.40}{0}
\node[purple] at (0,-1.6) {\Large$\cdot$};
\draw[-{Stealth[length=1.5mm]}, thin, purple] (0,-2.7) -- (0,-3.3);
\node[outl] at (0,-3.65) {No activation};
\end{scope}

% ---------- key ----------
\node[ago] at (-0.6,-4.45) {};
\node[anchor=west, font=\small] at (-0.3,-4.45) {agonist};
\antag{2.2}{-4.45}{0}
\node[anchor=west, font=\small] at (2.7,-4.45) {antagonist};
\fill[orange!35, draw=orange, thick] (5.0,-4.65) rectangle (5.4,-4.25);
\node[anchor=west, font=\small] at (5.5,-4.45) {receptor};
\end{tikzpicture}
