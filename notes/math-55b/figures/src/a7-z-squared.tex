  % left: square grid in [-1,1]^2
  \begin{scope}
    \foreach \c in {-1,-0.75,-0.5,-0.25,0,0.25,0.5,0.75,1} {
      \draw[red] (\c,-1) -- (\c,1);
      \draw[green] (-1,\c) -- (1,\c);
    }
    \draw[->, gray] (-1.3,0) -- (1.35,0);
    \draw[->, gray] (0,-1.3) -- (0,1.35);
    \node[below right, font=\small] at (1,0) {$1$};

    \node[above left, font=\small] at (0,1) {$1$};
    \node[below left, font=\small] at (0,-1) {$-1$};
  \end{scope}
  % arrow
  \draw[->, thick] (1.6,0.6) to[bend left=25] node[above] {$z\mapsto z^2$} (3.4,0.6);

  \begin{scope}[xshift=5cm, yscale=0.7]
    \foreach \c in {0.25,0.5,0.75,1} {

      \draw[green] plot[smooth, domain=-1:1, samples=41, variable=\t] ({(\t)*(\t) - (\c)*(\c)}, {2*(\c)*(\t)});
    }
    \draw[red] (-1,0) -- (0,0);
    \draw[green] (0,0) -- (1,0);
    \draw[->, gray] (-1.3,0) -- (1.35,0);
