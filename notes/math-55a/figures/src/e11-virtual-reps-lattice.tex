\begin{tikzpicture}[scale=0.8]
  % lattice points: actual representations (a_1, a_2 >= 0) in black, other virtual representations in red
  \foreach \x in {-3,...,3} {
    \foreach \y in {-2,...,2} {
      \pgfmathtruncatemacro{\isrep}{(\x>=0) && (\y>=0) ? 1 : 0}
      \ifnum\isrep=1
        \fill[black] (\x,\y) circle (2.2pt);
      \else
        \fill[red] (\x,\y) circle (2.2pt);
      \fi
    }
  }
  % boundary of the cone
  \draw[gray] (0,0) -- (3.5,0);
  \draw[gray] (0,0) -- (0,2.5);
  % generators
  \draw[-{Stealth[length=2.5mm]}, thick, blue] (0,0) -- (1,0) node[midway, below] {$[V_1]$};
  \draw[-{Stealth[length=2.5mm]}, thick, blue] (0,0) -- (0,1) node[midway, left] {$[V_2]$};
  % labels

  \draw (3.7,1) -- (3.85,1) node[right, align=left] {the cone of reps\\ inside $R(G)$};
  \node[red, align=center] at (-4.8,0.5) {virtual\\ reps};
\end{tikzpicture}
